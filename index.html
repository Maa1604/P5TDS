class P5Dataset(Dataset):
    def __init__(self, features, labels):
        assert features.shape[1] == 85
        assert len(features) == len(labels)
        # Guardamos los datos en esta instancia (self).
        self.features = features.astype(np.float32)
        self.labels = labels

    def __len__(self):
        # Número de ejemplos del dataset.
        return len(self.labels)

    def __getitem__(self, idx):
        # Devuelve el vector y la etiqueta de dataset[idx].
        return self.features[idx], self.labels[idx]

Capa nn.Module

from torch import nn

class Capa(nn.Module):
    def __init__(self, input_dim, out_dim):
        super().__init__()
        self.linear = nn.Linear(input_dim, out_dim)
        self.norm = nn.BatchNorm1d(out_dim)
        self.relu = nn.ReLU()

    def forward(self, x):
        x = self.linear(x)
        x = self.norm(x)
        return self.relu(x)

Red nn.Sequential

class Red(nn.Sequential):

Entrenar y validar una época

train_epoch Ajusta los pesos

def train_epoch(model, loader, loss_fn, optimizer):
    model.train()
    total_loss = correct = 0
    for x, y in loader:
        optimizer.zero_grad()
        scores = model(x)
        loss = loss_fn(scores, y)
        loss.backward()
        optimizer.step()
        total_loss += loss.item()
        correct += (scores.argmax(1) == y).sum().item()
    return total_loss / len(loader), correct / len(loader.dataset)

validate_epoch Mide el rendimiento

def validate_epoch(model, loader, loss_fn):
    model.eval()
    total_loss = correct = 0
    with torch.no_grad():
        for x, y in loader:
            scores = model(x)
            loss = loss_fn(scores, y)
            total_loss += loss.item()
            correct += (scores.argmax(1) == y).sum().item()
    return total_loss / len(loader), correct / len(loader.dataset)

def train(model, train_loader, val_loader,
          loss_fn, optimizer, num_epochs):
    history = []
    for _ in range(num_epochs):
        train_metrics = train_epoch(
            model, train_loader, loss_fn, optimizer
        )
        val_metrics = validate_epoch(
            model, val_loader, loss_fn
        )
        history.append((train_metrics, val_metrics))
    return history
